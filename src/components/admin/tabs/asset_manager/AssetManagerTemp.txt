  { id: 'identity', icon: '🏷️', label: '1. Định danh & Cờ' },

  discount: assetStore.assets.filter(a => a.hasDiscount).length,
  free: assetStore.assets.filter(a => a.isFree).length,
  upcoming: assetStore.upcomingAssets.length,
  roadmap: assetStore.roadmapAssets.length
}))

const filteredAssets = computed(() => {
  const q = searchQuery.value.toLowerCase().trim()
  return [...assetStore.assets]
    .filter(item => {
      const matchSearch = !q ||
        String(item.id).includes(q) ||
        (item.title || '').toLowerCase().includes(q) ||
        (item.slug || '').toLowerCase().includes(q) ||
        (item.tags || []).some(t => t.toLowerCase().includes(q))

      const matchCategory = categoryFilter.value === 'all' || item.categoryId === categoryFilter.value

      let matchStatus = true
      if (statusFilter.value === 'discount') matchStatus = item.hasDiscount
      else if (statusFilter.value === 'free') matchStatus = item.isFree
      else if (statusFilter.value === 'upcoming') matchStatus = Boolean(item.dates?.availableIn) || item.roadmap?.status === 'next'
      else if (statusFilter.value === 'roadmap') matchStatus = item.isInRoadmap

      return matchSearch && matchCategory && matchStatus
    })
    .sort((a, b) => {
      if (sortOrder.value === 'id_desc') return Number(b.id) - Number(a.id)
      if (sortOrder.value === 'updated_desc') return new Date(b.releaseDate) - new Date(a.releaseDate)
      if (sortOrder.value === 'price_desc') return (b.pricing?.price || 0) - (a.pricing?.price || 0)
      if (sortOrder.value === 'likes_desc') return (b.metrics?.likes || 0) - (a.metrics?.likes || 0)
      return Number(a.id) - Number(b.id)
    })
})

      .replace(/[\s_-]+/g, '-')
  }
}

const recalculateDiscountPrice = () => {
  const orig = Number(formData.value.pricing.originalPrice)
  const disc = Number(formData.value.pricing.discount)
  if (orig > 0 && disc > 0 && disc <= 100) {
    formData.value.pricing.price = Number((orig * (1 - disc / 100)).toFixed(2))
  }
}

const setAsFreePack = () => {
  formData.value.pricing.price = 0
  formData.value.pricing.originalPrice = null
  formData.value.pricing.discount = null
  formData.value.pricing.discountEndsIn = ''
}

const handleSaveAsset = async () => {
  if (!formData.value.title?.trim()) {
    alert('Vui lòng nhập Tên Asset Pack!')
    return
  }
  isSaving.value = true
  try {
    syncTextInputsToArrays()
    await assetStore.saveAssetToFirebase(formData.value)
    isModalOpen.value = false
  } catch (err) {
    alert('Lỗi khi lưu lên Firebase: ' + err.message)
  } finally {
    isSaving.value = false
  }
}

const handleCloneAsset = async (item) => {
  if (!confirm(`Bạn muốn nhân bản gói "${item.title}" thành một Pack mới?`)) return
  const cloned = JSON.parse(JSON.stringify(item))
  cloned.id = null
  cloned.title = `${item.title} (Copy)`
  cloned.slug = `${item.slug}-copy`
  await assetStore.saveAssetToFirebase(cloned)
}

// 3. Hàm xử lý khi Admin chọn file .js / .json từ máy tính để đẩy lên Firebase
const handleRestoreFileChange = async (event) => {
  const file = event.target.files?.[0]
  if (!file) return

  if (!confirm(`Bạn có chắc chắn muốn khôi phục dữ liệu Asset Pack từ file "${file.name}" lên Firebase không?`)) {
    event.target.value = ''
    return
  }

  isSaving.value = true
  try {
    const result = await restoreAssetsFromFile(file)
    alert(`Khôi phục thành công ${result.count} Asset Packs lên Firebase!`)
  } catch (err) {
    alert('Lỗi khi khôi phục file: ' + err.message)
  } finally {
    isSaving.value = false
    event.target.value = ''
  }
}
